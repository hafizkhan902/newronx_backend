1. API ENDPOINTS REFERENCE
================================================================================

🔗 BASE URL: /api/team-files

📤 UPLOAD ENDPOINTS
------------------

POST /api/team-files/upload
- Purpose: Direct file upload
- Content-Type: multipart/form-data
- Max Size: 50MB per file
- Body: FormData with 'file' + metadata
- Auth: Required (JWT Bearer token)

POST /api/team-files/upload-link
- Purpose: Link/URL upload
- Content-Type: application/json
- Body: {ideaId, url, title?, description?, tags?, category?}
- Auth: Required (JWT Bearer token)

📋 FILE RETRIEVAL ENDPOINTS
---------------------------

GET /api/team-files/idea/{ideaId}
- Purpose: Get all team files
- Query: page?, limit?, category?, fileType?, search?, sortBy?, sortOrder?
- Response: {files[], pagination{}}

GET /api/team-files/idea/{ideaId}/recent
- Purpose: Get recent files
- Query: limit? (default: 10)
- Response: files[]

GET /api/team-files/idea/{ideaId}/popular
- Purpose: Get most downloaded files
- Query: limit? (default: 10)
- Response: files[]

GET /api/team-files/idea/{ideaId}/categories
- Purpose: Get file categories with counts
- Response: [{category, count, totalSize}]

GET /api/team-files/idea/{ideaId}/stats
- Purpose: Get file statistics
- Response: {totalFiles, totalSize, totalDownloads, categorySummary}

GET /api/team-files/idea/{ideaId}/search
- Purpose: Search files
- Query: q (required), page?, limit?
- Response: {files[], pagination{}}

🔍 INDIVIDUAL FILE ENDPOINTS
----------------------------

GET /api/team-files/{fileId}
- Purpose: Get single file details
- Response: file object with full metadata

GET /api/team-files/{fileId}/download
- Purpose: Download/Access file
- Response: redirect for direct files, URL for links

PUT /api/team-files/{fileId}
- Purpose: Update file details
- Body: {description?, tags?, category?}
- Access: File uploader only

DELETE /api/team-files/{fileId}
- Purpose: Delete single file
- Access: File uploader or idea author

DELETE /api/team-files/bulk
- Purpose: Bulk delete files
- Body: {fileIds: []}
- Response: [{fileId, success, message/error}]

```javascript
// DirectFileUpload.jsx
import React, { useState, useRef } from 'react';

const DirectFileUpload = ({ ideaId, onUploadSuccess, onCancel }) => {
  const [file, setFile] = useState(null);
  const [description, setDescription] = useState('');
  const [tags, setTags] = useState('');
  const [category, setCategory] = useState('other');
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);
  const fileInputRef = useRef(null);

  const categories = [
    { value: 'document', label: '📄 Documents' },
    { value: 'image', label: '🖼️ Images' },
    { value: 'video', label: '🎥 Videos' },
    { value: 'audio', label: '🎵 Audio' },
    { value: 'archive', label: '📦 Archives' },
    { value: 'code', label: '💻 Code' },
    { value: 'design', label: '🎨 Design' },
    { value: 'other', label: '📋 Other' }
  ];

  const handleFileSelect = (event) => {
    const selectedFile = event.target.files[0];
    if (selectedFile) {
      // Validate file size (50MB limit)
      if (selectedFile.size > 50 * 1024 * 1024) {
        alert('File size exceeds 50MB limit');
        return;
      }
      setFile(selectedFile);
    }
  };

  const handleUpload = async () => {
    if (!file) {
      alert('Please select a file');
      return;
    }

    setUploading(true);
    setProgress(0);

    const formData = new FormData();
    formData.append('file', file);
    formData.append('ideaId', ideaId);
    formData.append('description', description);
    formData.append('tags', tags);
    formData.append('category', category);

    try {
      const response = await fetch('/api/team-files/upload', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        },
        body: formData
      });

      const result = await response.json();

      if (result.success) {
        onUploadSuccess(result.data);
        // Reset form
        setFile(null);
        setDescription('');
        setTags('');
        setCategory('other');
        fileInputRef.current.value = '';
      } else {
        throw new Error(result.message);
      }
    } catch (error) {
      console.error('Upload error:', error);
      alert(`Upload failed: ${error.message}`);
    } finally {
      setUploading(false);
      setProgress(0);
    }
  };

  const formatFileSize = (bytes) => {
    if (bytes === 0) return '0 B';
    const k = 1024;
    const sizes = ['B', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
  };

  return (
    <div className="direct-file-upload">
      <h3>📁 Direct File Upload</h3>
      
      {/* File Selection */}
      <div className="file-input-section">
        <input
          ref={fileInputRef}
          type="file"
          onChange={handleFileSelect}
          accept="*/*"
          style={{ display: 'none' }}
        />
        
        <div 
          className="file-drop-zone"
          onClick={() => fileInputRef.current?.click()}
          onDragOver={(e) => e.preventDefault()}
          onDrop={(e) => {
            e.preventDefault();
            const droppedFile = e.dataTransfer.files[0];
            if (droppedFile) {
              setFile(droppedFile);
            }
          }}
        >
          {file ? (
            <div className="file-selected">
              <div className="file-info">
                <span className="file-name">{file.name}</span>
                <span className="file-size">{formatFileSize(file.size)}</span>
              </div>
              <button onClick={(e) => {
                e.stopPropagation();
                setFile(null);
                fileInputRef.current.value = '';
              }}>
                ❌
              </button>
            </div>
          ) : (
            <div className="drop-zone-content">
              <div className="upload-icon">📤</div>
              <p>Click to select file or drag & drop</p>
              <span>Maximum file size: 50MB</span>
            </div>
          )}
        </div>
      </div>

      {/* File Metadata */}
      {file && (
        <div className="file-metadata">
          <div className="form-group">
            <label>Category</label>
            <select 
              value={category} 
              onChange={(e) => setCategory(e.target.value)}
            >
              {categories.map(cat => (
                <option key={cat.value} value={cat.value}>
                  {cat.label}
                </option>
              ))}
            </select>
          </div>

          <div className="form-group">
            <label>Description (Optional)</label>
            <textarea
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              placeholder="Describe what this file contains..."
              maxLength={500}
            />
          </div>

          <div className="form-group">
            <label>Tags (Optional)</label>
            <input
              type="text"
              value={tags}
              onChange={(e) => setTags(e.target.value)}
              placeholder="Enter tags separated by commas"
            />
          </div>
        </div>
      )}

      {/* Upload Progress */}
      {uploading && (
        <div className="upload-progress">
          <div className="progress-bar">
            <div 
              className="progress-fill" 
              style={{ width: `${progress}%` }}
            />
          </div>
          <span>Uploading... {progress}%</span>
        </div>
      )}

      {/* Action Buttons */}
      <div className="upload-actions">
        <button onClick={onCancel} disabled={uploading}>
          Cancel
        </button>
        <button 
          onClick={handleUpload} 
          disabled={!file || uploading}
          className="primary"
        >
          {uploading ? 'Uploading...' : 'Upload File'}
        </button>
      </div>
    </div>
  );
};

export default DirectFileUpload;
```

```javascript
// LinkUpload.jsx
import React, { useState } from 'react';

const LinkUpload = ({ ideaId, onUploadSuccess, onCancel }) => {
  const [url, setUrl] = useState('');
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [tags, setTags] = useState('');
  const [category, setCategory] = useState('other');
  const [uploading, setUploading] = useState(false);

  const categories = [
    { value: 'document', label: '📄 Documents' },
    { value: 'image', label: '🖼️ Images' },
    { value: 'video', label: '🎥 Videos' },
    { value: 'audio', label: '🎵 Audio' },
    { value: 'code', label: '💻 Code' },
    { value: 'design', label: '🎨 Design' },
    { value: 'other', label: '📋 Other' }
  ];

  const detectLinkType = (url) => {
    const urlLower = url.toLowerCase();
    
    if (urlLower.includes('youtube.com') || urlLower.includes('youtu.be')) {
      return { category: 'video', icon: '🎥', type: 'YouTube Video' };
    } else if (urlLower.includes('drive.google.com')) {
      return { category: 'document', icon: '📁', type: 'Google Drive' };
    } else if (urlLower.includes('dropbox.com')) {
      return { category: 'document', icon: '📦', type: 'Dropbox' };
    } else if (urlLower.includes('github.com')) {
      return { category: 'code', icon: '💻', type: 'GitHub Repository' };
    } else if (urlLower.includes('figma.com')) {
      return { category: 'design', icon: '🎨', type: 'Figma Design' };
    } else {
      return { category: 'other', icon: '🔗', type: 'External Link' };
    }
  };

  const linkInfo = url ? detectLinkType(url) : null;

  const validateUrl = (url) => {
    try {
      new URL(url);
      return true;
    } catch {
      return false;
    }
  };

  const handleUpload = async () => {
    if (!url.trim()) {
      alert('Please enter a URL');
      return;
    }

    if (!validateUrl(url)) {
      alert('Please enter a valid URL');
      return;
    }

    setUploading(true);

    try {
      const response = await fetch('/api/team-files/upload-link', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          ideaId,
          url: url.trim(),
          title: title.trim() || undefined,
          description: description.trim() || undefined,
          tags: tags.trim() || undefined,
          category: linkInfo?.category || category
        })
      });

      const result = await response.json();

      if (result.success) {
        onUploadSuccess(result.data);
        // Reset form
        setUrl('');
        setTitle('');
        setDescription('');
        setTags('');
        setCategory('other');
      } else {
        throw new Error(result.message);
      }
    } catch (error) {
      console.error('Link upload error:', error);
      alert(`Upload failed: ${error.message}`);
    } finally {
      setUploading(false);
    }
  };

  return (
    <div className="link-upload">
      <h3>🔗 Link Upload</h3>
      
      {/* URL Input */}
      <div className="form-group">
        <label>URL *</label>
        <input
          type="url"
          value={url}
          onChange={(e) => setUrl(e.target.value)}
          placeholder="https://example.com/file"
          required
        />
        
        {linkInfo && (
          <div className="link-preview">
            <span className="link-icon">{linkInfo.icon}</span>
            <span className="link-type">{linkInfo.type}</span>
          </div>
        )}
      </div>

      {/* Title Input */}
      <div className="form-group">
        <label>Title (Optional)</label>
        <input
          type="text"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          placeholder="Give this link a descriptive title"
        />
      </div>

      {/* Category Selection */}
      <div className="form-group">
        <label>Category</label>
        <select 
          value={linkInfo?.category || category} 
          onChange={(e) => setCategory(e.target.value)}
        >
          {categories.map(cat => (
            <option key={cat.value} value={cat.value}>
              {cat.label}
            </option>
          ))}
        </select>
      </div>

      {/* Description */}
      <div className="form-group">
        <label>Description (Optional)</label>
        <textarea
          value={description}
          onChange={(e) => setDescription(e.target.value)}
          placeholder="Describe what this link contains..."
          maxLength={500}
        />
      </div>

      {/* Tags */}
      <div className="form-group">
        <label>Tags (Optional)</label>
        <input
          type="text"
          value={tags}
          onChange={(e) => setTags(e.target.value)}
          placeholder="Enter tags separated by commas"
        />
      </div>

      {/* Action Buttons */}
      <div className="upload-actions">
        <button onClick={onCancel} disabled={uploading}>
          Cancel
        </button>
        <button 
          onClick={handleUpload} 
          disabled={!url.trim() || !validateUrl(url) || uploading}
          className="primary"
        >
          {uploading ? 'Adding Link...' : 'Add Link'}
        </button>
      </div>
    </div>
  );
};

export default LinkUpload;
```

```javascript
// TeamFileList.jsx
import React, { useState, useEffect } from 'react';

const TeamFileList = ({ ideaId }) => {
  const [files, setFiles] = useState([]);
  const [loading, setLoading] = useState(true);
  const [page, setPage] = useState(1);
  const [pagination, setPagination] = useState({});
  const [filters, setFilters] = useState({
    category: '',
    search: '',
    sortBy: 'createdAt',
    sortOrder: -1
  });

  const fetchFiles = async (pageNum = 1, newFilters = filters) => {
    try {
      setLoading(true);
      const queryParams = new URLSearchParams({
        page: pageNum,
        limit: 20,
        ...newFilters
      });

      const response = await fetch(
        `/api/team-files/idea/${ideaId}?${queryParams}`,
        {
          headers: {
            'Authorization': `Bearer ${localStorage.getItem('token')}`
          }
        }
      );

      const result = await response.json();

      if (result.success) {
        if (pageNum === 1) {
          setFiles(result.data.files);
        } else {
          setFiles(prev => [...prev, ...result.data.files]);
        }
        setPagination(result.data.pagination);
      }
    } catch (error) {
      console.error('Error fetching files:', error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchFiles();
  }, [ideaId]);

  const handleFilterChange = (newFilters) => {
    setFilters(newFilters);
    setPage(1);
    fetchFiles(1, newFilters);
  };

  const handleLoadMore = () => {
    const nextPage = page + 1;
    setPage(nextPage);
    fetchFiles(nextPage);
  };

  const handleFileUpdate = (updatedFile) => {
    setFiles(prev => prev.map(file => 
      file._id === updatedFile._id ? updatedFile : file
    ));
  };

  const handleFileDelete = (deletedFileId) => {
    setFiles(prev => prev.filter(file => file._id !== deletedFileId));
  };

  return (
    <div className="team-file-list">
      {/* File Filters */}
      <FileFilters 
        filters={filters} 
        onFilterChange={handleFilterChange}
        ideaId={ideaId}
      />

      {/* File Grid */}
      <div className="files-grid">
        {files.map(file => (
          <FileCard
            key={file._id}
            file={file}
            onUpdate={handleFileUpdate}
            onDelete={handleFileDelete}
          />
        ))}
      </div>

      {/* Load More */}
      {pagination.hasNext && (
        <div className="load-more">
          <button onClick={handleLoadMore} disabled={loading}>
            {loading ? 'Loading...' : 'Load More Files'}
          </button>
        </div>
      )}

      {/* Empty State */}
      {!loading && files.length === 0 && (
        <div className="empty-state">
          <div className="empty-icon">📁</div>
          <h3>No files yet</h3>
          <p>Upload your first file to get started with team collaboration!</p>
        </div>
      )}
    </div>
  );
};

export default TeamFileList;
```

```javascript
// FileCard.jsx
import React, { useState } from 'react';

const FileCard = ({ file, onUpdate, onDelete }) => {
  const [showActions, setShowActions] = useState(false);
  const [downloading, setDownloading] = useState(false);

  const getFileIcon = (file) => {
    if (file.uploadMethod === 'link') {
      const url = file.url.toLowerCase();
      if (url.includes('youtube.com') || url.includes('youtu.be')) return '🎥';
      if (url.includes('drive.google.com')) return '📁';
      if (url.includes('dropbox.com')) return '📦';
      if (url.includes('github.com')) return '💻';
      if (url.includes('figma.com')) return '🎨';
      return '🔗';
    }

    const ext = file.fileExtension;
    if (['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext)) return '🖼️';
    if (['pdf', 'doc', 'docx'].includes(ext)) return '📄';
    if (['mp4', 'avi', 'mov'].includes(ext)) return '🎥';
    if (['mp3', 'wav', 'flac'].includes(ext)) return '🎵';
    if (['zip', 'rar', '7z'].includes(ext)) return '📦';
    if (['js', 'html', 'css', 'py'].includes(ext)) return '💻';
    return '📋';
  };

  const getCategoryBadge = (category) => {
    const badges = {
      document: { icon: '📄', color: '#3498db' },
      image: { icon: '🖼️', color: '#e74c3c' },
      video: { icon: '🎥', color: '#9b59b6' },
      audio: { icon: '🎵', color: '#f39c12' },
      archive: { icon: '📦', color: '#95a5a6' },
      code: { icon: '💻', color: '#2ecc71' },
      design: { icon: '🎨', color: '#e67e22' },
      other: { icon: '📋', color: '#34495e' }
    };
    
    const badge = badges[category] || badges.other;
    return (
      <span 
        className="category-badge" 
        style={{ backgroundColor: badge.color }}
      >
        {badge.icon} {category}
      </span>
    );
  };

  const handleDownload = async () => {
    try {
      setDownloading(true);
      
      if (file.uploadMethod === 'link') {
        // For links, open in new tab
        window.open(file.url, '_blank');
      } else {
        // For direct uploads, use download endpoint
        const response = await fetch(`/api/team-files/${file._id}/download`, {
          headers: {
            'Authorization': `Bearer ${localStorage.getItem('token')}`
          }
        });
        
        if (response.ok) {
          window.location.href = response.url;
        }
      }
    } catch (error) {
      console.error('Download error:', error);
      alert('Failed to download file');
    } finally {
      setDownloading(false);
    }
  };

  const handleDelete = async () => {
    if (!window.confirm('Are you sure you want to delete this file?')) {
      return;
    }

    try {
      const response = await fetch(`/api/team-files/${file._id}`, {
        method: 'DELETE',
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        }
      });

      const result = await response.json();

      if (result.success) {
        onDelete(file._id);
      } else {
        throw new Error(result.message);
      }
    } catch (error) {
      console.error('Delete error:', error);
      alert(`Failed to delete file: ${error.message}`);
    }
  };

  return (
    <div 
      className="file-card"
      onMouseEnter={() => setShowActions(true)}
      onMouseLeave={() => setShowActions(false)}
    >
      {/* File Preview */}
      <div className="file-preview">
        {file.isImage && file.uploadMethod === 'direct' ? (
          <img src={file.url} alt={file.originalName} />
        ) : (
          <div className="file-icon">
            {getFileIcon(file)}
          </div>
        )}
        
        {/* Upload Method Badge */}
        <div className="upload-method-badge">
          {file.uploadMethod === 'direct' ? '📁' : '🔗'}
        </div>
      </div>

      {/* File Info */}
      <div className="file-info">
        <h4 className="file-name" title={file.originalName}>
          {file.originalName}
        </h4>
        
        {file.description && (
          <p className="file-description">{file.description}</p>
        )}

        <div className="file-meta">
          {getCategoryBadge(file.category)}
          
          <div className="file-details">
            <span className="file-size">
              {file.uploadMethod === 'direct' ? file.fileSizeFormatted : 'Link'}
            </span>
            <span className="upload-date">
              {new Date(file.createdAt).toLocaleDateString()}
            </span>
          </div>
        </div>

        {/* Uploader Info */}
        <div className="uploader-info">
          <img 
            src={file.uploader.avatar || '/default-avatar.png'} 
            alt={file.uploader.fullName}
            className="uploader-avatar"
          />
          <span className="uploader-name">{file.uploader.firstName}</span>
        </div>

        {/* Tags */}
        {file.tags && file.tags.length > 0 && (
          <div className="file-tags">
            {file.tags.map((tag, index) => (
              <span key={index} className="tag">#{tag}</span>
            ))}
          </div>
        )}

        {/* Download Count */}
        <div className="download-info">
          <span>📥 {file.downloadCount} downloads</span>
        </div>
      </div>

      {/* Action Buttons */}
      {showActions && (
        <div className="file-actions">
          <button 
            onClick={handleDownload}
            disabled={downloading}
            className="action-btn download"
            title={file.uploadMethod === 'link' ? 'Open Link' : 'Download'}
          >
            {downloading ? '⏳' : (file.uploadMethod === 'link' ? '🔗' : '📥')}
          </button>
          
          <button 
            onClick={() => onUpdate(file)}
            className="action-btn edit"
            title="Edit Details"
          >
            ✏️
          </button>
          
          <button 
            onClick={handleDelete}
            className="action-btn delete"
            title="Delete File"
          >
            🗑️
          </button>
        </div>
      )}
    </div>
  );
};

export default FileCard;
```

🔧 MAIN TEAM FILES COMPONENT

```javascript
// TeamFiles.jsx
import React, { useState } from 'react';
import UploadMethodSelector from './UploadMethodSelector';
import DirectFileUpload from './DirectFileUpload';
import LinkUpload from './LinkUpload';
import TeamFileList from './TeamFileList';
import FileStats from './FileStats';

const TeamFiles = ({ ideaId }) => {
  const [showUpload, setShowUpload] = useState(false);
  const [uploadMethod, setUploadMethod] = useState(null);
  const [refreshTrigger, setRefreshTrigger] = useState(0);

  const handleUploadSuccess = (uploadedFile) => {
    console.log('File uploaded successfully:', uploadedFile);
    setShowUpload(false);
    setUploadMethod(null);
    setRefreshTrigger(prev => prev + 1); // Trigger file list refresh
  };

  const handleCancelUpload = () => {
    setShowUpload(false);
    setUploadMethod(null);
  };

  return (
    <div className="team-files">
      <div className="team-files-header">
        <h2>📁 Team Files</h2>
        <button 
          onClick={() => setShowUpload(true)}
          className="upload-btn primary"
        >
          + Upload File
        </button>
      </div>

      {/* File Statistics */}
      <FileStats ideaId={ideaId} refreshTrigger={refreshTrigger} />

      {/* Upload Modal */}
      {showUpload && (
        <div className="upload-modal">
          <div className="modal-content">
            {!uploadMethod ? (
              <UploadMethodSelector 
                onMethodSelect={setUploadMethod}
                selectedMethod={uploadMethod}
              />
            ) : uploadMethod === 'direct' ? (
              <DirectFileUpload
                ideaId={ideaId}
                onUploadSuccess={handleUploadSuccess}
                onCancel={handleCancelUpload}
              />
            ) : (
              <LinkUpload
                ideaId={ideaId}
                onUploadSuccess={handleUploadSuccess}
                onCancel={handleCancelUpload}
              />
            )}
          </div>
          
          <div 
            className="modal-backdrop"
            onClick={handleCancelUpload}
          />
        </div>
      )}

      {/* File List */}
      <TeamFileList ideaId={ideaId} key={refreshTrigger} />
    </div>
  );
};

export default TeamFiles;
```

```typescript
// types/teamFiles.ts
export interface TeamFile {
  _id: string;
  ideaId: string;
  uploadedBy: string;
  filename: string;
  originalName: string;
  description?: string;
  uploadMethod: 'direct' | 'link';
  fileType: string;
  fileSize: number;
  url: string;
  category: FileCategory;
  tags: string[];
  isPublic: boolean;
  downloadCount: number;
  lastDownloaded?: Date;
  metadata: {
    width?: number;
    height?: number;
    duration?: number;
    pages?: number;
    format?: string;
  };
  version: number;
  parentFileId?: string;
  isActive: boolean;
  createdAt: Date;
  updatedAt: Date;
  
  // Virtuals
  fileSizeFormatted: string;
  fileExtension: string;
  isImage: boolean;
  isDocument: boolean;
  
  // Populated fields
  uploader: {
    _id: string;
    firstName: string;
    fullName: string;
    avatar?: string;
  };
}

export type FileCategory = 
  | 'document' 
  | 'image' 
  | 'video' 
  | 'audio' 
  | 'archive' 
  | 'code' 
  | 'design' 
  | 'other';

export interface FileUploadData {
  ideaId: string;
  file: File;
  description?: string;
  tags?: string[];
  category?: FileCategory;
}

export interface LinkUploadData {
  ideaId: string;
  url: string;
  title?: string;
  description?: string;
  tags?: string[];
  category?: FileCategory;
}

export interface FileFilters {
  category?: FileCategory;
  fileType?: string;
  uploadedBy?: string;
  search?: string;
  sortBy?: string;
  sortOrder?: number;
}

export interface FilePagination {
  currentPage: number;
  totalPages: number;
  totalFiles: number;
  hasNext: boolean;
  hasPrev: boolean;
}

export interface FileStats {
  totalFiles: number;
  totalSize: number;
  totalDownloads: number;
  categorySummary: Record<FileCategory, number>;
  byUploadMethod: Array<{
    method: 'direct' | 'link';
    count: number;
  }>;
}
```

8. QUICK START GUIDE
================================================================================

1. **Install Dependencies** (if not already installed):
   ```bash
   npm install multer cloudinary
   ```

2. **Start Backend Server**:
   ```bash
   npm start
   ```

3. **Test Upload Endpoints**:
   ```bash
   # Direct upload
   curl -X POST http://localhost:2000/api/team-files/upload \
     -H "Authorization: Bearer YOUR_TOKEN" \
     -F "ideaId=YOUR_IDEA_ID" \
     -F "file=@test-file.pdf" \
     -F "description=Test file" \
     -F "category=document"

   # Link upload
   curl -X POST http://localhost:2000/api/team-files/upload-link \
     -H "Authorization: Bearer YOUR_TOKEN" \
     -H "Content-Type: application/json" \
     -d '{
       "ideaId": "YOUR_IDEA_ID",
       "url": "https://github.com/example/repo",
       "title": "Project Repository",
       "category": "code"
     }'
   ```

4. **Integrate Frontend Components**:
   - Copy components from this guide
   - Add to your team management section
   - Customize styling to match your design
   - Test upload and download functionality
